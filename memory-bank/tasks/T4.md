## Description

Switch the chat panel's AI calls from the current blocking `.invoke()` to LangChain's `.stream()`. Chunks are yielded as an async iterable and piped to the React chat component for progressive display. An `AbortController` enables mid-stream cancellation.

This task is self-contained and has no dependencies on other new tasks. It can be completed first and tested via the existing inline tooltip path or a simple test harness before the chat panel (T1) is built.

## Implementation Sketch

```typescript
// In ChatApiManager
async *streamChat(
  messages: BaseMessage[],
  signal: AbortSignal
): AsyncIterable<string> {
  if (!this.chatClient) throw new Error("Chat client not initialised");
  const stream = await this.chatClient.stream(messages, { signal });
  for await (const chunk of stream) {
    yield chunk.content.toString();
  }
}

// In Chat component
const controller = useRef<AbortController | null>(null);

async function handleSubmit(userMessage: string) {
  controller.current = new AbortController();
  let fullText = "";
  try {
    for await (const chunk of chatApi.streamChat(messages, controller.current.signal)) {
      fullText += chunk;
      setCurrentAiMessage(fullText);
    }
    appendMessage({ role: "assistant", content: fullText });
  } catch (e) {
    if (e.name === "AbortError") {
      appendMessage({ role: "assistant", content: fullText + " [stopped]" });
    } else {
      appendMessage({ role: "assistant", content: "Error: " + e.message, isError: true });
    }
  } finally {
    setCurrentAiMessage("");
    controller.current = null;
  }
}
```
